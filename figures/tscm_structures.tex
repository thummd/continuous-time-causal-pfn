% Node styles. Filled = treatment / outcome; outlined = observed
% covariate / mediator; dashed = unobserved confounder.
\tikzset{
    obs/.style    = {circle, draw, minimum size=8mm, inner sep=0pt, font=\small},
    treat/.style  = {circle, draw, fill=blue!18, minimum size=8mm, inner sep=0pt, font=\small\bfseries},
    out/.style    = {circle, draw, fill=red!18, minimum size=8mm, inner sep=0pt, font=\small\bfseries},
    hid/.style    = {circle, draw, dashed, minimum size=8mm, inner sep=0pt, font=\small},
    arr/.style    = {-{Stealth[length=2.4mm]}, thick},
    title/.style  = {font=\small\bfseries},
}

\newcommand{\panel}[3]{%
  % #1 = panel title
  % #2 = identification class label (small italics underneath)
  % #3 = TikZ body
  \begin{tikzpicture}[baseline=(current bounding box.north)]
    \node[title] (T) at (0, 1.7) {#1};
    \node[font=\footnotesize\itshape] at (0, 1.35) {#2};
    \begin{scope}[yshift=-0.1cm]
        #3
    \end{scope}
  \end{tikzpicture}%
}
