% Continuous-time "collapsed" view (cf. Boeken & Mooij 2026, Fig. 1a):
% one node per variable, a self-loop for the SDE's own-value dependence
% (-theta_v X_v), and u->v wherever u enters v's drift. do(a) severs a's
% incoming edges, self-loop included (red dashed).
\begingroup
\tikzset{
  ct/.style   = {circle, draw=blue!55!black, fill=blue!15, minimum size=6.4mm,
                 inner sep=0pt, font=\small},
  cint/.style = {circle, draw=red!70!black, thick, fill=orange!75, minimum size=6.4mm,
                 inner sep=0pt, font=\small},
  cout/.style = {circle, draw=orange!85!black, thick, fill=white, minimum size=6.4mm,
                 inner sep=0pt, font=\small},
  chid/.style = {circle, draw=black!70, dashed, fill=white, minimum size=6.4mm,
                 inner sep=0pt, font=\small},
  ar/.style   = {-{Stealth[length=1.9mm]}, blue!55!black, semithick},
  cut/.style  = {-{Stealth[length=1.9mm]}, red!75!black, thick, dashed},
  slL/.style  = {ar,  out=160, in=200, looseness=7},   % self-loop, left
  slR/.style  = {ar,  out=-20, in=20,  looseness=7},   % self-loop, right
  slU/.style  = {ar,  out=70,  in=110, looseness=7},   % self-loop, up
  slD/.style  = {ar,  out=250, in=290, looseness=7},   % self-loop, down
  slLcut/.style={cut, out=160, in=200, looseness=7},   % severed self-loop
  ttl/.style  = {font=\small\bfseries},
  fam/.style  = {font=\footnotesize\itshape, text=black!70},
}

  \end{tikzpicture}}
